function record(name, pass, detail = {}) {

async function request(name, method, url, cookie, body, expected, assertion) {
  try {
    const form = body instanceof FormData;
    const r = await fetch(`http://localhost:3100${url}`, {
      method, redirect: 'manual', signal: AbortSignal.timeout(30000),
      headers: { ...(cookie ? { cookie } : {}), ...(body && !form ? { 'content-type': 'application/json' } : {}) },
      body: body === undefined ? undefined : form ? body : JSON.stringify(body),
    });
    const text = await r.text();
    let data; try { data = JSON.parse(text); } catch { data = null; }
    const cookies = r.headers.getSetCookie().map(s => s.split(';')[0]).join('; ');
    const allowed = Array.isArray(expected) ? expected : [expected];
    record(name, allowed.includes(r.status) && (!assertion || assertion(data, r)), { method, url, status: r.status, expected: allowed });
    return { data, status: r.status, cookie: cookies, headers: r.headers };
  } catch (e) { record(name, false, { method, url, error: e.name }); return {}; }
}
async function login(name, username) {
  return (await request(name, 'POST', '/api/auth/login', null, { username, password }, 200, d => !!d.role && !d.user?.password && !d.club?.password)).cookie;
}
async function nextStudent() {
  for (;;) { const id = String(99000000 + crypto.randomInt(999999)); if (!await db.user.findUnique({ where: { studentId: id } })) { owned.studentIds.push(id); return id; } }
}
async function seedRole(role) {
  const studentId = await nextStudent();
  const u = await db.user.create({ data: { studentId, email: `${tag}_${role}@example.invalid`, role, password: await bcrypt.hash(password, 10) } });
  owned.userIds.push(u.id); return { ...u, cookie: await login(`login ${role}`, studentId) };
}
function walk(dir) { return fs.readdirSync(dir, { withFileTypes: true }).flatMap(e => e.isDirectory() ? walk(path.join(dir, e.name)) : [path.join(dir, e.name)]); }

async function cleanup() {
  for (const item of owned.uploads) {
    try { const { createClient } = require('@supabase/supabase-js'); const s = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_SERVICE_ROLE_KEY); const r = await s.storage.from(item.bucket).remove([item.path]); out.cleanup.storage = !r.error; } catch { out.cleanup.storage = false; }
  }
  try {
    // Exact generated IDs only; cascade removes children of this run's fixtures.
    const userWhere = { OR: [{ id: { in: owned.userIds } }, { studentId: { in: owned.studentIds } }] };
    await db.user.deleteMany({ where: userWhere });
    await db.competition.deleteMany({ where: { id: { in: owned.competitionIds } } });
    await db.club.deleteMany({ where: { id: { in: owned.clubIds } } });
    out.cleanup.remainingUsers = await db.user.count({ where: userWhere });
    out.cleanup.remainingCompetitions = await db.competition.count({ where: { id: { in: owned.competitionIds } } });
    out.cleanup.remainingClubs = await db.club.count({ where: { id: { in: owned.clubIds } } });
  } catch (e) { out.cleanup.databaseError = e.code || e.name; }
  await db.$disconnect(); await pool.end();
  out.finished = new Date().toISOString();
  out.summary = { total: out.cases.length, passed: out.cases.filter(c => c.pass).length, failed: out.cases.filter(c => !c.pass).length };
  fs.writeFileSync('test-results/integration.json', JSON.stringify(out, null, 2));
  console.log(JSON.stringify({ summary: out.summary, cleanup: out.cleanup }));
}
main().catch(e => { record('test fixture/runtime blocker', false, { error: e.code || e.name, detail: e.code ? undefined : e.message }); }).finally(cleanup);
